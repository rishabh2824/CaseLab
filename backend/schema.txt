-- Case Lab Schema (SQLite)


-- 1. CASES
CREATE TABLE cases (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  case_name TEXT NOT NULL,
  initial_brief TEXT NOT NULL,
  common_information TEXT,
  simulation_duration INTEGER,
  non_referred INTEGER NOT NULL CHECK (non_referred >= 1),
  access_code TEXT,
  owner_admin_id TEXT REFERENCES admins(id));


-- 2. PERSONAS
CREATE TABLE personas (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  case_id TEXT NOT NULL,
  name TEXT NOT NULL DEFAULT '',
  role TEXT NOT NULL DEFAULT '',
  known_facts TEXT,
  personality_traits TEXT,
  availability_duration INTEGER, profile_photo_file_id TEXT
REFERENCES files(id) ON DELETE SET NULL,
  FOREIGN KEY (case_id) REFERENCES cases(id) ON DELETE CASCADE
);


-- 3. REFERRAL RELATIONSHIP
CREATE TABLE persona_referrals (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  case_id TEXT NOT NULL,
  parent_persona_id TEXT NOT NULL,
  referred_persona_id TEXT NOT NULL,
  condition_trigger TEXT,
  FOREIGN KEY (case_id) REFERENCES cases(id) ON DELETE CASCADE,
  FOREIGN KEY (parent_persona_id) REFERENCES personas(id) ON DELETE CASCADE,
  FOREIGN KEY (referred_persona_id) REFERENCES personas(id) ON DELETE CASCADE
);


-- 4. FILES (Metadata for your DO Spaces)
CREATE TABLE files (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  bucket TEXT NOT NULL,
  object_key TEXT NOT NULL,
  file_name TEXT NOT NULL,
  content_type TEXT,
  upload_status TEXT NOT NULL DEFAULT 'uploaded',
  UNIQUE(bucket, object_key)
);


-- 5. PERSONA FILES
CREATE TABLE persona_files (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  persona_id TEXT NOT NULL,
  file_id TEXT,
  share_conditions TEXT,
  perceived_contents TEXT,
  FOREIGN KEY (persona_id) REFERENCES personas(id) ON DELETE CASCADE,
  FOREIGN KEY (file_id) REFERENCES files(id) ON DELETE SET NULL
);


-- 6. INDEXES
CREATE UNIQUE INDEX idx_cases_access_code_upper
  ON cases (upper(access_code))
  WHERE access_code IS NOT NULL AND access_code != '';


-- 7. ADMINS
CREATE TABLE admins (
  id TEXT PRIMARY KEY DEFAULT (lower(hex(randomblob(16)))),
  email TEXT NOT NULL UNIQUE,
  name TEXT,
  role INTEGER NOT NULL CHECK (role IN (1, 2))  -- 1 = super admin, 2 = admin
);


-- 8. SIMULATION RUNS
CREATE TABLE simulation_runs (
  run_id TEXT PRIMARY KEY,
  expires_at REAL NOT NULL,
  data TEXT NOT NULL,
  version INTEGER NOT NULL DEFAULT 0);


CREATE INDEX idx_simulation_runs_expires_at ON simulation_runs (expires_at);

-- 9. RATE LIMITS
CREATE TABLE rate_limits (
  key TEXT PRIMARY KEY,
  window_start REAL NOT NULL,
  count INTEGER NOT NULL
);
